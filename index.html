📋 Supabase SQL Schema (คลิกเพื่อดูคำสั่งสร้างตารางใน Supabase)

คัดลอกคำสั่ง SQL นี้ไปวางใน SQL Editor บน Supabase dashboard เพื่อสร้างตารางและเปิด Realtime:

-- 1. ตารางคำสั่งซื้อและการจัดส่ง (Orders)
CREATE TABLE IF NOT EXISTS public.orders (
    id TEXT PRIMARY KEY,
    so TEXT NOT NULL,
    type TEXT NOT NULL DEFAULT 'SALE',
    customer TEXT DEFAULT '',
    dest TEXT DEFAULT '',
    dn TEXT DEFAULT '',
    received TEXT DEFAULT '',
    ship TEXT DEFAULT '',
    remark TEXT DEFAULT '',
    created TIMESTAMPTZ DEFAULT NOW(),
    updated TIMESTAMPTZ DEFAULT NOW()
);

-- 2. ตารางป้ายสีกำกับวัน (Day Markers)
CREATE TABLE IF NOT EXISTS public.day_marks (
    date TEXT PRIMARY KEY,
    color TEXT NOT NULL,
    label TEXT DEFAULT '',
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- 3. ตารางวันหยุด (Holidays)
CREATE TABLE IF NOT EXISTS public.holidays (
    date TEXT PRIMARY KEY,
    name TEXT NOT NULL,
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- 4. ตารางชื่อหมวดหมู่สี (Mark Names)
CREATE TABLE IF NOT EXISTS public.mark_names (
    color TEXT PRIMARY KEY,
    label TEXT NOT NULL
);

-- 5. ตารางบันทึกประวัติการเปลี่ยนแปลง (Audit Log)
CREATE TABLE IF NOT EXISTS public.calendar_logs (
    id BIGSERIAL PRIMARY KEY,
    ts TIMESTAMPTZ DEFAULT NOW(),
    action TEXT NOT NULL,
    ref TEXT DEFAULT '',
    before TEXT DEFAULT '',
    after TEXT DEFAULT ''
);

-- เปิดใช้งาน Realtime
ALTER PUBLICATION supabase_realtime ADD TABLE public.orders;
ALTER PUBLICATION supabase_realtime ADD TABLE public.day_marks;
ALTER PUBLICATION supabase_realtime ADD TABLE public.holidays;
ALTER PUBLICATION supabase_realtime ADD TABLE public.mark_names;
ALTER PUBLICATION supabase_realtime ADD TABLE public.calendar_logs;

-- เปิดใช้งาน Row Level Security (RLS)
ALTER TABLE public.orders ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.day_marks ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.holidays ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.mark_names ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.calendar_logs ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Allow public read orders" ON public.orders FOR SELECT USING (true);
CREATE POLICY "Allow public all orders" ON public.orders FOR ALL USING (true);

CREATE POLICY "Allow public read day_marks" ON public.day_marks FOR SELECT USING (true);
CREATE POLICY "Allow public all day_marks" ON public.day_marks FOR ALL USING (true);

CREATE POLICY "Allow public read holidays" ON public.holidays FOR SELECT USING (true);
CREATE POLICY "Allow public all holidays" ON public.holidays FOR ALL USING (true);

CREATE POLICY "Allow public read mark_names" ON public.mark_names FOR SELECT USING (true);
CREATE POLICY "Allow public all mark_names" ON public.mark_names FOR ALL USING (true);

CREATE POLICY "Allow public read logs" ON public.calendar_logs FOR SELECT USING (true);
CREATE POLICY "Allow public insert logs" ON public.calendar_logs FOR INSERT WITH CHECK (true);
