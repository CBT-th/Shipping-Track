📋 Supabase SQL Schema (คลิกเพื่อดูคำสั่งสร้างตารางใน Supabase)

คัดลอกคำสั่ง SQL นี้ไปวางใน SQL Editor บน Supabase dashboard เพื่อสร้างตารางและเปิด Realtime:

-- ==========================================================
-- 🏢 Enterprise Shipping Calendar Database Schema & RLS Setup
-- ==========================================================

-- 1. ตารางคำสั่งซื้อและการจัดส่ง (Orders)
CREATE TABLE IF NOT EXISTS public.orders (
    id TEXT PRIMARY KEY,
    so TEXT NOT NULL,
    type TEXT NOT NULL DEFAULT 'SALE',
    customer TEXT DEFAULT '',
    dest TEXT DEFAULT '',
    dn TEXT DEFAULT '',
    received TEXT DEFAULT '',
    ship TEXT DEFAULT '',
    remark TEXT DEFAULT '',
    created TIMESTAMPTZ DEFAULT NOW(),
    updated TIMESTAMPTZ DEFAULT NOW()
);

-- 2. ตารางป้ายสีกำกับวัน (Day Markers)
CREATE TABLE IF NOT EXISTS public.day_marks (
    date TEXT PRIMARY KEY,
    color TEXT NOT NULL,
    label TEXT DEFAULT '',
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- 3. ตารางวันหยุด (Holidays)
CREATE TABLE IF NOT EXISTS public.holidays (
    date TEXT PRIMARY KEY,
    name TEXT NOT NULL,
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- 4. ตารางชื่อหมวดหมู่สี (Mark Names)
CREATE TABLE IF NOT EXISTS public.mark_names (
    color TEXT PRIMARY KEY,
    label TEXT NOT NULL
);

-- 5. ตารางบันทึกประวัติการเปลี่ยนแปลง (Audit Log)
CREATE TABLE IF NOT EXISTS public.calendar_logs (
    id BIGSERIAL PRIMARY KEY,
    ts TIMESTAMPTZ DEFAULT NOW(),
    action TEXT NOT NULL,
    ref TEXT DEFAULT '',
    before TEXT DEFAULT '',
    after TEXT DEFAULT ''
);

-- 6. ตารางบันทึกการเข้าดูและล็อกอิน (Access Log)
CREATE TABLE IF NOT EXISTS public.access_logs (
    id BIGSERIAL PRIMARY KEY,
    ts TIMESTAMPTZ DEFAULT NOW(),
    user_name TEXT DEFAULT '',
    role TEXT DEFAULT '',
    department TEXT DEFAULT '',
    device TEXT DEFAULT '',
    action TEXT DEFAULT '',
    details TEXT DEFAULT ''
);

-- 7. ตารางพนักงานและสิทธิ์การเข้าใช้งาน (Users)
CREATE TABLE IF NOT EXISTS public.users (
    email TEXT PRIMARY KEY,
    name TEXT NOT NULL,
    dept TEXT DEFAULT '',
    password TEXT DEFAULT 'CBTUDON22',
    role TEXT NOT NULL DEFAULT 'SALES',
    status TEXT NOT NULL DEFAULT 'active',
    mobile_readonly BOOLEAN DEFAULT true,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- เปิดใช้งาน Realtime Replication สำหรับทุกตาราง (ใช้ SET TABLE ป้องกัน Error 42710)
ALTER PUBLICATION supabase_realtime SET TABLE 
    public.orders,
    public.day_marks,
    public.holidays,
    public.mark_names,
    public.calendar_logs,
    public.access_logs,
    public.users;

-- ==========================================================
-- 🔒 Enterprise Row Level Security (RLS) Policy Setup
-- ==========================================================
ALTER TABLE public.orders ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.day_marks ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.holidays ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.mark_names ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.calendar_logs ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.access_logs ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.users ENABLE ROW LEVEL SECURITY;

-- ล้าง Policy เดิมเพื่อป้องกันการซ้ำซ้อน
DROP POLICY IF EXISTS "orders_read_policy" ON public.orders;
DROP POLICY IF EXISTS "orders_write_policy" ON public.orders;
DROP POLICY IF EXISTS "day_marks_read_policy" ON public.day_marks;
DROP POLICY IF EXISTS "day_marks_write_policy" ON public.day_marks;
DROP POLICY IF EXISTS "holidays_read_policy" ON public.holidays;
DROP POLICY IF EXISTS "holidays_write_policy" ON public.holidays;
DROP POLICY IF EXISTS "mark_names_read_policy" ON public.mark_names;
DROP POLICY IF EXISTS "mark_names_write_policy" ON public.mark_names;
DROP POLICY IF EXISTS "logs_read_policy" ON public.calendar_logs;
DROP POLICY IF EXISTS "logs_insert_policy" ON public.calendar_logs;
DROP POLICY IF EXISTS "access_logs_read_policy" ON public.access_logs;
DROP POLICY IF EXISTS "access_logs_insert_policy" ON public.access_logs;
DROP POLICY IF EXISTS "users_read_policy" ON public.users;
DROP POLICY IF EXISTS "users_write_policy" ON public.users;

-- สร้าง Enterprise Access Policies
CREATE POLICY "orders_read_policy" ON public.orders FOR SELECT USING (true);
CREATE POLICY "orders_write_policy" ON public.orders FOR ALL USING (true);

CREATE POLICY "day_marks_read_policy" ON public.day_marks FOR SELECT USING (true);
CREATE POLICY "day_marks_write_policy" ON public.day_marks FOR ALL USING (true);

CREATE POLICY "holidays_read_policy" ON public.holidays FOR SELECT USING (true);
CREATE POLICY "holidays_write_policy" ON public.holidays FOR ALL USING (true);

CREATE POLICY "mark_names_read_policy" ON public.mark_names FOR SELECT USING (true);
CREATE POLICY "mark_names_write_policy" ON public.mark_names FOR ALL USING (true);

CREATE POLICY "logs_read_policy" ON public.calendar_logs FOR SELECT USING (true);
CREATE POLICY "logs_insert_policy" ON public.calendar_logs FOR INSERT WITH CHECK (true);

CREATE POLICY "access_logs_read_policy" ON public.access_logs FOR SELECT USING (true);
CREATE POLICY "access_logs_insert_policy" ON public.access_logs FOR INSERT WITH CHECK (true);

CREATE POLICY "users_read_policy" ON public.users FOR SELECT USING (true);
CREATE POLICY "users_write_policy" ON public.users FOR ALL USING (true);
